## Usage

```tsx
import { Link } from '@codeui/kit';

export function FooterLinks() {
  return (
    <>
      <Link href="https://codeimage.dev" target="_blank">
        Open CodeImage
      </Link>
      <Link href="https://github.com/riccardoperra/codeui" target="_blank" variant="underline">
        View source
      </Link>
      <Link href="#" disabled>
        Export history
      </Link>
    </>
  );
}
```

## API

`Link` renders an `a` element. It wraps Kobalte's `Link`, so Kobalte's props and the anchor attributes such as `href` and `target` pass through.
Use `as` to render a router link instead.
