## Usage

```tsx
import { createSignal } from 'solid-js';
import { Select } from '@codeui/kit';

const fonts = ['Jetbrains Mono', 'Fira Code', 'Source Code Pro', 'Cascadia Code'];

export function FontPicker() {
  const [font, setFont] = createSignal('Jetbrains Mono');
  return (
    <Select
      aria-label="Font"
      label="Font"
      placeholder="Choose a font"
      options={fonts}
      value={font()}
      onChange={value => value && setFont(value)}
    />
  );
}
```

## API

`Select` wraps Kobalte's `Select` and renders the label, the trigger, the messages, and a portaled list.
Kobalte's root props such as `multiple`, `disabled`, `optionValue`, `optionTextValue`, and `optionDisabled` pass through.
`aria-label` is required and names the trigger.

## Object options

`createSelectOptions` maps a list of objects to options, using `key` for the label and `valueKey` for the value.
Spread its `props()` into the select, and convert between your value and the option with `optionFromValue`.

<SolidDemo name="select-options" source="hidden" />

```tsx
import { createSignal } from 'solid-js';
import { createSelectOptions, Select } from '@codeui/kit';

const themes = [
  { name: 'Dracula', id: 'dracula' },
  { name: 'One Dark', id: 'one-dark' },
  { name: 'Night Owl', id: 'night-owl' },
];

export function ThemePicker() {
  const [theme, setTheme] = createSignal('dracula');
  const themeOptions = createSelectOptions(themes, { key: 'name', valueKey: 'id' });
  return (
    <Select
      aria-label="Theme"
      label="Theme"
      placeholder="Choose a theme"
      options={themeOptions.options()}
      {...themeOptions.props()}
      value={themeOptions.optionFromValue(theme())}
      onChange={option => option && setTheme(option.value)}
    />
  );
}
```
